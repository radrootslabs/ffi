cmake_minimum_required(VERSION 3.24)
project(RadrootsCppConsumer LANGUAGES CXX)
find_package(RadrootsCpp CONFIG REQUIRED)
set(RADROOTS_LINKAGE shared CACHE STRING "C linkage")
option(RADROOTS_NO_EXCEPTIONS "Disable consumer exceptions" OFF)
add_executable(consumer main.cpp)
if(RADROOTS_LINKAGE STREQUAL "static")
    target_link_libraries(consumer PRIVATE Radroots::cpp_static)
else()
    target_link_libraries(consumer PRIVATE Radroots::cpp)
endif()
target_compile_options(consumer PRIVATE -Wall -Wextra -Werror)
if(RADROOTS_NO_EXCEPTIONS)
    target_compile_options(consumer PRIVATE -fno-exceptions)
endif()
